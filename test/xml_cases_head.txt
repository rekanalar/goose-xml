import std;
import xml from "..";

fn hex(out: u8[>..]&, s: const u8[:]) {
    let digits = "0123456789abcdef";
    var i = 0;
    while i < s.len {
        out.push(digits[(s[i] >> 4) as i64]);
        out.push(digits[(s[i] & 15) as i64]);
        i += 1;
    }
}

fn is_decl(name: const u8[:]) -> bool {
    if name == "xmlns" { return true; }
    if name.len > 6 && name[0..6] == "xmlns:" { return true; }
    return false;
}

recursive fn walk(out: u8[>..]&, doc: xml::Doc&, id: i32) {
    let n = doc.nodes[id];
    if n.kind == xml::KIND_TEXT {
        var dec: u8[>..] = [];
        xml::decode_entities(dec, n.content);
        out.append("|");
        hex(out, dec[..]);
        return;
    }
    if n.kind == xml::KIND_CDATA {
        var cd: u8[>..] = [];
        var k = 0;
        while k < n.content.len { k = xml::put_literal(cd, n.content, k, false); }
        out.append("|");
        hex(out, cd[..]);
        return;
    }
    if n.kind != xml::KIND_ELEMENT { return; }
    out.append("E{");
    if n.uri.len > 0 {
        out.append("{");
        out.append(n.uri);
        out.append("}");
    }
    out.append(n.local);
    out.append(";");
    var k = 0;
    var first = true;
    while k < n.attr_count {
        let a = doc.attrs[n.attr_start + k];
        k += 1;
        if is_decl(a.name) { continue; }
        if !first { out.append(";"); }
        first = false;
        if a.uri.len > 0 {
            out.append("{");
            out.append(a.uri);
            out.append("}");
        }
        out.append(a.local);
        out.append("=");
        var dec: u8[>..] = [];
        xml::decode_attr(dec, a.value);
        hex(out, dec[..]);
    }
    out.append("}");
    var c = n.first_child;
    while c >= 0 {
        walk(out, doc, c);
        c = doc.nodes[c].next_sibling;
    }
    out.append("/");
}

// merge joins adjacent text runs and drops empty runs, as the oracle does.
fn merge(out: u8[>..]&, s: const u8[:]) {
    var i = 0;
    while i < s.len {
        if s[i] != '|' {
            out.push(s[i]);
            i += 1;
            continue;
        }
        var run: u8[>..] = [];
        while i < s.len && s[i] == '|' {
            i += 1;
            while i < s.len && ((s[i] >= '0' && s[i] <= '9') || (s[i] >= 'a' && s[i] <= 'f')) {
                run.push(s[i]);
                i += 1;
            }
        }
        if run.len > 0 {
            out.push('|');
            out.append(run[..]);
        }
    }
}

fn render_of(out: u8[>..]&, s: const u8[:]) -> bool {
    var nodes: xml::Node[>..] = [];
    var attrs: xml::Attr[>..] = [];
    let doc, ok = xml::parse(s, nodes, attrs);
    if !ok { return false; }
    var raw: u8[>..] = [];
    walk(raw, doc, doc.root);
    merge(out, raw[..]);
    return true;
}

fn expect_ok(id: i64, s: const u8[:], want: const u8[:]) {
    var got: u8[>..] = [];
    if !render_of(got, s) {
        print("case ", id, " refused: ", s);
        assert(false);
    }
    if got[..] != want {
        print("case ", id, " differs: ", s, "\n  want ", want, "\n  got  ", got[..]);
        assert(false);
    }
    // The writer output must parse again to the same tree.
    var nodes: xml::Node[>..] = [];
    var attrs: xml::Attr[>..] = [];
    let doc, ok = xml::parse(s, nodes, attrs);
    var built: u8[>..] = [];
    xml::build(built, doc, doc.root);
    var again: u8[>..] = [];
    if !render_of(again, built[..]) || again[..] != want {
        print("case ", id, " round trip: ", built[..]);
        assert(false);
    }
}

fn expect_bad(id: i64, s: const u8[:]) {

    }
}

// tag_text writes the tag of the node as ElementTree writes it.
fn tag_text(out: u8[>..]&, doc: xml::Doc&, id: i32) {
    let node = doc.nodes[id];
    if node.uri.len > 0 {
        out.push('{');
        out.append(node.uri);
        out.push('}');
    }
    out.append(node.local);
}

// expect_path checks the elements that match the path.
fn expect_path(id: i64, s: const u8[:], path: const u8[:], want: const u8[:]) {
    var nodes: xml::Node[>..] = [];
    var attrs: xml::Attr[>..] = [];
    let doc, ok = xml::parse(s, nodes, attrs);
    if !ok { print("path case ", id, " refused the document"); assert(false); }
    var ids: i32[>..] = [];
    let n = xml::findall_path(doc, doc.root, path, ids);
    if n < 0 { print("path case ", id, " refused the path: ", path); assert(false); }
    var got: u8[>..] = [];
    var i = 0;
    while i < ids.len {
        if i > 0 { got.push('|'); }
        tag_text(got, doc, ids[i]);
        i += 1;
    }
    if got[..] != want {
        print("path case ", id, " differs. path: ", path, "\n  want ", want, "\n  got  ", got[..]);
        assert(false);
    }
}

// expect_attrs checks the names and the values of the attributes.
fn expect_attrs(id: i64, s: const u8[:], want_keys: const u8[:], want_items: const u8[:]) {
    var nodes: xml::Node[>..] = [];
    var attrs: xml::Attr[>..] = [];
    let doc, ok = xml::parse(s, nodes, attrs);
    if !ok { print("attr case ", id, " refused the document"); assert(false); }
    var kbuf: u8[>..] = [];
    var names: const u8[:][>..] = [];
    xml::keys(doc, doc.root, kbuf, names);
    var got_keys: u8[>..] = [];
    var i = 0;
    while i < names.len {
        if i > 0 { got_keys.push('|'); }
        got_keys.append(names[i]);
        i += 1;
    }
    if got_keys[..] != want_keys {
        print("attr case ", id, " differs.\n  want ", want_keys, "\n  got  ", got_keys[..]);
        assert(false);
    }
    var buf: u8[>..] = [];
    var pairs: xml::Pair[>..] = [];
    xml::items(doc, doc.root, buf, pairs);
    var got_items: u8[>..] = [];
    var j = 0;
    while j < pairs.len {
        if j > 0 { got_items.push('|'); }
        got_items.append(pairs[j].name);
        got_items.push('=');
        got_items.append(pairs[j].value);
        j += 1;
    }
    if got_items[..] != want_items {
        print("attr case ", id, " differs.\n  want ", want_items, "\n  got  ", got_items[..]);
        assert(false);
    }
}

// expect_tostring checks the text that the writer makes.
fn expect_tostring(id: i64, s: const u8[:], short_empty: bool, decl: bool, want: const u8[:]) {
    var nodes: xml::Node[>..] = [];
    var attrs: xml::Attr[>..] = [];
    let doc, ok = xml::parse(s, nodes, attrs);
    if !ok { print("write case ", id, " refused the document"); assert(false); }
    var got: u8[>..] = [];
    let opts = xml::WriteOpts { xml_declaration: if decl { xml::DECL_YES as i32 } else { xml::DECL_AUTO as i32 }, short_empty_elements: short_empty, method: xml::METHOD_XML as i32, default_namespace: "" };
    if !xml::tostring(got, doc, doc.root, opts) { print("write case ", id, " failed"); assert(false); }
    if got[..] != want {
        print("write case ", id, " differs. doc: ", s, "\n  want ", want, "\n  got  ", got[..]);
        assert(false);
    }
}

// expect_indent checks the text after indent.
fn expect_indent(id: i64, s: const u8[:], space: const u8[:], level: i64, want: const u8[:]) {
    var nodes: xml::Node[>..] = [];
    var attrs: xml::Attr[>..] = [];
    var text: u8[>..] = [];
    let doc, ok = xml::parse(s, nodes, attrs);
    if !ok { print("indent case ", id, " refused the document"); assert(false); }
    let root = doc.root;
    if !xml::indent(nodes, attrs, text, root, space, level) { print("indent case ", id, " failed"); assert(false); }
    let again = xml::make_doc(nodes, attrs, root);
    var got: u8[>..] = [];
    if !xml::tostring(got, again, root, xml::write_opts()) { print("indent case ", id, " failed to write"); assert(false); }
    if got[..] != want {
        print("indent case ", id, " differs. doc: ", s, "\n  want ", want, "\n  got  ", got[..]);
        assert(false);
    }
}

// expect_c14n checks the canonical form. An empty want with ok false means
// that the call must fail.
fn expect_c14n(id: i64, s: const u8[:], wc: bool, st: bool, rw: bool,
        et: const u8[:][:], ea: const u8[:][:], qt: const u8[:][:], qa: const u8[:][:],
        ok: bool, want: const u8[:]) {
    var nodes: xml::Node[>..] = [];
    var attrs: xml::Attr[>..] = [];
    let doc, pok = xml::parse(s, nodes, attrs);
    if !pok { print("c14n case ", id, " refused the document"); assert(false); }
    var got: u8[>..] = [];
    let opts = xml::C14nOpts { with_comments: wc, strip_text: st, rewrite_prefixes: rw };
    let res = xml::canonicalize(got, doc, opts, et, ea, qt, qa);
    if res != ok { print("c14n case ", id, " result ", res, " for ", s); assert(false); }
    if ok && got[..] != want {
        print("c14n case ", id, " differs. doc: ", s, "\n  want ", want, "\n  got  ", got[..]);
        assert(false);
    }
}

// render_enc parses bytes in any supported encoding and renders the tree.
fn render_enc(out: u8[>..]&, s: const u8[:]) -> bool {
    var buf: u8[>..] = [];
    var nodes: xml::Node[>..] = [];
    var attrs: xml::Attr[>..] = [];
    let doc, ok = xml::parse_bytes(s, buf, nodes, attrs);
    if !ok { return false; }
    var raw: u8[>..] = [];
    walk(raw, doc, doc.root);
    merge(out, raw[..]);
    return true;
}

// expect_enc_ok checks a document in an encoding.
fn expect_enc_ok(id: i64, s: const u8[:], want: const u8[:]) {
    var got: u8[>..] = [];
    if !render_enc(got, s) { print("encoding case ", id, " refused: ", s); assert(false); }
    if got[..] != want {
        print("encoding case ", id, " differs: ", s, "\n  want ", want, "\n  got  ", got[..]);
        assert(false);
    }
}

// expect_enc_bad checks that a document in an encoding is refused.
fn expect_enc_bad(id: i64, s: const u8[:]) {
    var got: u8[>..] = [];
    if render_enc(got, s) { print("encoding case ", id, " accepted: ", s); assert(false); }
}

// expect_events checks the events of iterparse and the tree that it builds.
fn expect_events(id: i64, s: const u8[:], want_events: const u8[:], want_tree: const u8[:]) {
    var nodes: xml::Node[>..] = [];
    var attrs: xml::Attr[>..] = [];
    var text: u8[>..] = [];
    var ev: u8[>..] = [];
    let doc, ok = xml::iterparse(s, true, nodes, attrs, text) { kind, eid, a, b =>
        if kind != xml::TEV_TEXT {
            if ev.len > 0 { ev.push(' '); }
            if kind == xml::TEV_START { ev.append("S "); }
            else if kind == xml::TEV_END { ev.append("E "); }
            else if kind == xml::TEV_ATTR { ev.append("A "); }
            else if kind == xml::TEV_START_NS { ev.append("N "); }
            else if kind == xml::TEV_END_NS { ev.append("n "); }
            else if kind == xml::TEV_COMMENT { ev.append("C "); }
            else { ev.append("P "); }
            hex(ev, if kind == xml::TEV_END_NS { "" } else { a });
            if kind == xml::TEV_ATTR || kind == xml::TEV_START_NS || kind == xml::TEV_PI {
                ev.push(' ');
                hex(ev, b);
            }
        }
    };
    if !ok { print("event case ", id, " refused: ", s); assert(false); }
    if ev[..] != want_events {
        print("event case ", id, " events differ: ", s, "\n  want ", want_events, "\n  got  ", ev[..]);
        assert(false);
    }
    var raw: u8[>..] = [];
    walk(raw, doc, doc.root);
    var got: u8[>..] = [];
    merge(got, raw[..]);
    if got[..] != want_tree {
        print("event case ", id, " tree differs: ", s, "\n  want ", want_tree, "\n  got  ", got[..]);
        assert(false);
    }
}
